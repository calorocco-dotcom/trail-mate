cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(trailmate_navigation LANGUAGES CXX)
endif()
set(CMAKE_CXX_STANDARD 17)
set(REPO "${CMAKE_CURRENT_LIST_DIR}/../..")
include("${REPO}/cmake/TrailMateTmap.cmake")
trailmate_add_tmap()
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_target_property(sdk_source tmap SOURCE_DIR)
add_executable(navigation_calculation test_calculation.cpp)
target_include_directories(navigation_calculation PRIVATE "${REPO}/modules/ui_map_runtime/include")
target_link_libraries(navigation_calculation PRIVATE tmap::tmap)
if(MSVC)
    target_compile_options(navigation_calculation PRIVATE /W4 /UNDEBUG)
else()
    target_compile_options(navigation_calculation PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME navigation_fixture COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/make_fixture.py"
    "${sdk_source}/tools" "${CMAKE_CURRENT_BINARY_DIR}/navigation.tmap")
set_tests_properties(navigation_fixture PROPERTIES FIXTURES_SETUP navigation_map)
add_test(NAME navigation_calculation COMMAND navigation_calculation "${CMAKE_CURRENT_BINARY_DIR}/navigation.tmap")
set_tests_properties(navigation_calculation PROPERTIES FIXTURES_REQUIRED navigation_map FIXTURES_SETUP calculated_navigation)
add_test(NAME navigation_saved_kml COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/verify_saved_kml.py"
    "${CMAKE_CURRENT_BINARY_DIR}/navigation.tmap.kml")
set_tests_properties(navigation_saved_kml PROPERTIES FIXTURES_REQUIRED calculated_navigation)
