# Compile the real production mode parser; do not mirror it in a test fake.
set(SD_RUNTIME_SOURCE "${REPO}/platform/esp/arduino_common/src/storage/sd_card_runtime.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SD_RUNTIME_SOURCE}")
file(READ "${SD_RUNTIME_SOURCE}" SD_RUNTIME_CONTENT)
string(FIND "${SD_RUNTIME_CONTENT}" "oflag_t sdfat_open_flags(" SD_MODE_START)
string(FIND "${SD_RUNTIME_CONTENT}" "// SdFat::end()" SD_MODE_END)
if(SD_MODE_START LESS 0 OR SD_MODE_END LESS SD_MODE_START)
    message(FATAL_ERROR "Cannot locate production SD file-mode parser")
endif()
math(EXPR SD_MODE_LENGTH "${SD_MODE_END}-${SD_MODE_START}")
string(SUBSTRING "${SD_RUNTIME_CONTENT}" ${SD_MODE_START} ${SD_MODE_LENGTH} SD_MODE_CONTENT)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sd_open_flags_actual.inc" "${SD_MODE_CONTENT}")
add_executable(navigation_sd_route_modes test_sd_route_modes.cpp)
target_include_directories(navigation_sd_route_modes PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
    target_compile_options(navigation_sd_route_modes PRIVATE /W4 /UNDEBUG)
else()
    target_compile_options(navigation_sd_route_modes PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME navigation_sd_route_modes COMMAND navigation_sd_route_modes)

# The native IDF backend implements the same port with its own mode parser.
set(IDF_SD_SOURCE "${REPO}/platform/esp/idf_common/src/sd_card_runtime_sdfat_adapter.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${IDF_SD_SOURCE}")
file(READ "${IDF_SD_SOURCE}" IDF_SD_CONTENT)
string(FIND "${IDF_SD_CONTENT}" "oflag_t sdfat_open_flags(" IDF_MODE_START)
string(FIND "${IDF_SD_CONTENT}" "class SdmmcBlockDevice" IDF_MODE_END)
if(IDF_MODE_START LESS 0 OR IDF_MODE_END LESS IDF_MODE_START)
    message(FATAL_ERROR "Cannot locate production IDF SD file-mode parser")
endif()
math(EXPR IDF_MODE_LENGTH "${IDF_MODE_END}-${IDF_MODE_START}")
string(SUBSTRING "${IDF_SD_CONTENT}" ${IDF_MODE_START} ${IDF_MODE_LENGTH} IDF_MODE_CONTENT)
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/sd_open_flags_actual.inc" "namespace idf {\n${IDF_MODE_CONTENT}\n}\n")
