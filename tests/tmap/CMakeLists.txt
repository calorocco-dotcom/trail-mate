cmake_minimum_required(VERSION 3.20)
project(tmap_reader_contract LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(REPO "${CMAKE_CURRENT_LIST_DIR}/../..")
include("${REPO}/cmake/TrailMateTmap.cmake")
trailmate_add_tmap()
enable_testing()
get_target_property(trailmate_tmap_source tmap SOURCE_DIR)
add_test(NAME tmap_platform_link_isolation
    COMMAND "${CMAKE_COMMAND}"
        -S "${CMAKE_CURRENT_LIST_DIR}/export_context"
        -B "${CMAKE_CURRENT_BINARY_DIR}/export-context"
        -G "${CMAKE_GENERATOR}"
        "-DCMAKE_CXX_COMPILER=${CMAKE_CXX_COMPILER}"
        "-DCMAKE_MAKE_PROGRAM=${CMAKE_MAKE_PROGRAM}"
        "-DFETCHCONTENT_SOURCE_DIR_TRAILMATE_TMAP=${trailmate_tmap_source}")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME tmap_dependency_contract COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_LIST_DIR}/test_dependency_contract.py")
add_executable(tmap_released_sdk_smoke released_sdk_smoke.cpp)
target_link_libraries(tmap_released_sdk_smoke PRIVATE tmap::tmap)
add_test(NAME tmap_released_sdk_smoke COMMAND tmap_released_sdk_smoke)
add_executable(tmap_administrative_path test_administrative_path.cpp)
target_link_libraries(tmap_administrative_path PRIVATE tmap::tmap)
if(MSVC)
    target_compile_options(tmap_administrative_path PRIVATE /UNDEBUG)
else()
    target_compile_options(tmap_administrative_path PRIVATE -UNDEBUG)
endif()
add_test(NAME tmap_administrative_path COMMAND tmap_administrative_path)
add_executable(tmap_probe probe.cpp)
target_link_libraries(tmap_probe PRIVATE tmap::tmap)
if(MSVC)
    target_compile_options(tmap_probe PRIVATE /utf-8 /W4)
else()
    target_compile_options(tmap_probe PRIVATE -Wall -Wextra -Werror)
    target_compile_definitions(tmap_probe PRIVATE _FILE_OFFSET_BITS=64)
endif()
set(TRAILMATE_TMAP_WORLD_MAP "" CACHE FILEPATH "Optional real world v1 map")
set(TRAILMATE_TMAP_CHINA_MAP "" CACHE FILEPATH "Optional real China v1 map")
if(TRAILMATE_TMAP_WORLD_MAP AND TRAILMATE_TMAP_CHINA_MAP)
    add_test(NAME tmap_world COMMAND tmap_probe "${TRAILMATE_TMAP_WORLD_MAP}" 7 116.4 39.9)
    add_test(NAME tmap_china COMMAND tmap_probe "${TRAILMATE_TMAP_CHINA_MAP}" 12 102.7 25.04)
    add_test(NAME tmap_source_contract COMMAND tmap_source_probe "${TRAILMATE_TMAP_WORLD_MAP}" "${TRAILMATE_TMAP_CHINA_MAP}")
endif()
add_executable(tmap_source_probe source_probe.cpp
    ${REPO}/modules/ui_map_runtime/src/map_tiles/tmap_map_tile_source.cpp)
target_include_directories(tmap_source_probe PRIVATE
    ${REPO}/modules/ui_map_runtime/include ${REPO}/modules/ui_presentation/include)
target_link_libraries(tmap_source_probe PRIVATE tmap::tmap)
if(MSVC)
    target_compile_options(tmap_source_probe PRIVATE /utf-8 /W4 /UNDEBUG)
else()
    target_compile_options(tmap_source_probe PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_compile_definitions(tmap_source_probe PRIVATE _FILE_OFFSET_BITS=64)
endif()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../navigation" navigation)
