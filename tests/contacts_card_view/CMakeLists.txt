cmake_minimum_required(VERSION 3.20)
project(contacts_card_view_contract LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LVGL_DIR "" CACHE PATH "LVGL source directory")
set(REPO "${CMAKE_CURRENT_LIST_DIR}/../..")
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(${LVGL_DIR} lvgl)
add_executable(contacts_card_view_contract test_view.cpp)
target_include_directories(contacts_card_view_contract PRIVATE ${REPO}/modules/ui_shared/include)
target_link_libraries(contacts_card_view_contract PRIVATE lvgl)
enable_testing()
add_test(NAME contacts_card_view_contract COMMAND contacts_card_view_contract)
