function(extract source start end output)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
    file(READ "${source}" text)
    string(FIND "${text}" "${start}" first)
    string(FIND "${text}" "${end}" last)
    if(first LESS 0 OR last LESS first)
        message(FATAL_ERROR "Missing production boundary: ${start}")
    endif()
    math(EXPR length "${last}-${first}")
    string(SUBSTRING "${text}" ${first} ${length} result)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${result}")
endfunction()
