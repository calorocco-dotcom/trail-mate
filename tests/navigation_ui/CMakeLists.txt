cmake_minimum_required(VERSION 3.20)
project(navigation_dialog_contract LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(REPO "${CMAKE_CURRENT_LIST_DIR}/../..")
set(LVGL_DIR "" CACHE PATH "LVGL 9.4 source directory")
set(LVGL_LIBRARY "" CACHE FILEPATH "Optional compatible prebuilt LVGL library")
set(LV_BUILD_CONF_PATH "${REPO}/tests/contacts_card_view/lv_conf.h" CACHE PATH "" FORCE)
if(LVGL_LIBRARY)
    add_library(lvgl STATIC IMPORTED)
    set_target_properties(lvgl PROPERTIES IMPORTED_LOCATION "${LVGL_LIBRARY}")
    target_include_directories(lvgl INTERFACE "${LVGL_DIR}" "${REPO}/tests/contacts_card_view")
    target_compile_definitions(lvgl INTERFACE LV_CONF_INCLUDE_SIMPLE)
else()
    set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
    set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    add_subdirectory("${LVGL_DIR}" lvgl)
endif()
add_executable(navigation_dialog_contract test_dialog.cpp
    ${REPO}/modules/ui_shared/src/ui/screens/tracker/tracker_navigation_dialog.cpp
    ${REPO}/modules/ui_shared/src/ui/components/two_pane_styles.cpp)
target_include_directories(navigation_dialog_contract PRIVATE
    ${REPO}/modules/ui_shared/include ${REPO}/modules/core_sys/include ${REPO}/modules/core_gps/include)
# Select the device port declarations while exercising UI with host port fakes.
target_compile_definitions(navigation_dialog_contract PRIVATE ARDUINO_ARCH_ESP32)
target_link_libraries(navigation_dialog_contract PRIVATE lvgl)
if(MSVC)
    target_compile_options(navigation_dialog_contract PRIVATE /utf-8 /UNDEBUG)
else()
    target_compile_options(navigation_dialog_contract PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME navigation_dialog_contract COMMAND navigation_dialog_contract)
